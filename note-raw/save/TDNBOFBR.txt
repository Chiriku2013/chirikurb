_G.autoQuestEnabled = false
local isFirstTimeLoad = true 
local currentNotiInstance = nil

local function tpInFront(targetObj)
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") or not targetObj then return end
    local pivot = targetObj:GetPivot()
    -- Giữ khoảng cách 3.5 studs phía trước để bấm Prompt dễ dàng
    char.HumanoidRootPart.CFrame = pivot * CFrame.new(0, 0, -3.5) * CFrame.Angles(0, math.pi, 0)
end

-- Đọc Notification Quest hiện tại
local function getActiveQuestNoti()
    local guiFolder = LocalPlayer:FindFirstChild("PlayerGui")
        and LocalPlayer.PlayerGui:FindFirstChild("GUI")
        and LocalPlayer.PlayerGui.GUI:FindFirstChild("Frames")
        and LocalPlayer.PlayerGui.GUI.Frames:FindFirstChild("Notifications")

    if guiFolder then
        local children = guiFolder:GetChildren()
        for i = #children, 1, -1 do
            local child = children[i]
            if (child.Name == "PersistentNotification" or child.Name == "ActiveNotification") and child:IsA("TextLabel") then
                local cleanText = string.gsub(child.Text or "", "<[^>]->", "")
                local lowerText = string.lower(cleanText)
                
                local rarityStr = string.match(lowerText, "bring me a (.-) brainrot!")
                               or string.match(lowerText, "i need a (.-) brainrot!")
                
                if rarityStr then
                    for exactName, folderID in pairs(rarityToFolder) do
                        if string.lower(exactName) == rarityStr then
                            return exactName, folderID, child
                        end
                    end
                end
            end
        end
    end
    return nil, nil, nil
end

-- Tìm Item có Mutation cao nhất
local function getBestItem(folderNum, targetRarity)
    local specificFolder = workspace:FindFirstChild("ItemSpawns") and workspace.ItemSpawns:FindFirstChild(tostring(folderNum))
    if not specificFolder then return nil end

    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end

    local bestItem = nil
    local bestRank = -1
    local shortestDist = math.huge

    for _, obj in ipairs(specificFolder:GetChildren()) do
        if obj.Name == "SpawnedItem" then
            local rarity = obj:GetAttribute("Rarity")
            if rarity == targetRarity then
                local mutation = obj:GetAttribute("Mutation") or "Normal"
                local rank = mutationRanks[mutation] or 1
                local dist = (obj:GetPivot().Position - hrp.Position).Magnitude

                if rank > bestRank then
                    bestRank = rank
                    shortestDist = dist
                    bestItem = obj
                elseif rank == bestRank and dist < shortestDist then
                    shortestDist = dist
                    bestItem = obj
                end
            end
        end
    end
    return bestItem
end

---------------------------------------------------------
-- VÒNG LẶP AUTO QUEST CHUẨN XÁC
---------------------------------------------------------
local function startAutoQuestLoop()
    task.spawn(function()
        local npc = workspace:FindFirstChild("Quest") and workspace.Quest:FindFirstChild("NPC")
        if not npc then return end

        while _G.autoQuestEnabled do
            -----------------------------------------------------
            -- BƯỚC 1: TELEPORT TỚI NPC VÀ ĐỜI NHẬN QUEST MỚI
            -----------------------------------------------------
            local targetRarity, targetFolder, notiObj = nil, nil, nil
            
            while _G.autoQuestEnabled do
                tpInFront(npc)
                targetRarity, targetFolder, notiObj = getActiveQuestNoti()
                
                -- Nếu phát hiện Notification MỚI xuất hiện -> Nhận Quest thành công!
                if targetRarity and notiObj ~= currentNotiInstance then
                    currentNotiInstance = notiObj
                    break
                end
                task.wait(0.2)
            end
            
            if not _G.autoQuestEnabled then break end

            -----------------------------------------------------
            -- BƯỚC 2: ĐI ĐẾN ITEM VÀ ĐỜI NGƯỜI CHƠI NHẶT
            -----------------------------------------------------
            local targetItem = nil
            
            -- Đợi tới khi tìm thấy Item trong map
            while _G.autoQuestEnabled do
                targetItem = getBestItem(targetFolder, targetRarity)
                if targetItem then break end
                
                -- Nếu chưa có Item thì Teleport đợi ở khu vực đó
                task.wait(0.3)
            end

            -- Giữ vị trí tại Item cho tới khi Item bị biến mất (Người chơi đã nhặt xong)
            while _G.autoQuestEnabled and targetItem and targetItem.Parent do
                tpInFront(targetItem)
                task.wait(0.1)
            end

            if not _G.autoQuestEnabled then break end

            -----------------------------------------------------
            -- BƯỚC 3: TELEPORT VỀ NPC ĐỂ TRẢ QUEST
            -----------------------------------------------------
            -- Teleport tới NPC và đợi Notification Quest cũ biến mất (Xác nhận đã trả Quest)
            while _G.autoQuestEnabled do
                tpInFront(npc)
                
                local activeRarity, _, activeNoti = getActiveQuestNoti()
                -- Nếu Notification cũ đã mất hoặc chuyển sang Noti mới -> Trả thành công!
                if activeNoti ~= currentNotiInstance then
                    break
                end
                task.wait(0.2)
            end
            
            task.wait(0.3) -- Delay nhỏ để ổn định luồng trước khi sang quest tiếp theo
        end
    end)
end

autoQuestBtn.MouseButton1Click:Connect(function()
    _G.autoQuestEnabled = not _G.autoQuestEnabled
    
    if _G.autoQuestEnabled then
        autoQuestBtn.Text = "Auto Quest: ON"
        autoQuestBtn.BackgroundColor3 = Color3.fromRGB(50, 180, 50)
        currentNotiInstance = nil 

        task.spawn(function()
            -- Chạy Tour Map lần đầu (Load Chunks)
            if isFirstTimeLoad then
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    for i = 2, #teleLocations do
                        if not _G.autoQuestEnabled then break end
                        char.HumanoidRootPart.CFrame = CFrame.new(teleLocations[i].pos)
                        task.wait(0.35)
                    end
                end
                isFirstTimeLoad = false
            end
            
            if _G.autoQuestEnabled then
                startAutoQuestLoop()
            end
        end)
    else
        autoQuestBtn.Text = "Auto Quest: OFF"
        autoQuestBtn.BackgroundColor3 = Color3.fromRGB(180, 50, 50)
    end
end)
