-- Hàm hỗ trợ Instant Trigger ProximityPrompt không dùng VIM
local function interactPrompt(parentObject)
    if not parentObject then return end
    local prompt = parentObject:FindFirstChildOfClass("ProximityPrompt") or parentObject:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt then
        prompt.HoldDuration = 0
        if fireproximityprompt then
            fireproximityprompt(prompt)
        end
    end
end

-- Hàm dịch chuyển
local function teleportTo(targetPos)
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if hrp then
        hrp.CFrame = CFrame.new(targetPos + Vector3.new(0, 3, 0))
    end
end

-- Logic Auto Quest Core
local autoQuestEnabled = false
local isQuestRunning = false

-- Lấy tên Rarity và Số Folder tương ứng từ Notification
local function getNotificationInfo()
    local playerGui = LocalPlayer:WaitForChild("PlayerGui")
    local notificationLabel = playerGui:FindFirstChild("GUI") 
        and playerGui.GUI:FindFirstChild("Notifications") 
        and playerGui.GUI.Notifications:FindFirstChild("PersistentNotification")
        
    if notificationLabel and notificationLabel.Text ~= "" then
        local text = notificationLabel.Text
        for key, folderNum in pairs(rarityToFolder) do
            if string.find(text, key) then
                return key, folderNum -- Trả về tên Rarity và Số Folder
            end
        end
    end
    return nil, nil
end

-- Tìm SpawnedItem gần nhất khớp với Attribute Rarity
local function getNearestMatchingItem(targetRarity, folderNum)
    local itemFolder = workspace:FindFirstChild("ItemSpawns") and workspace.ItemSpawns:FindFirstChild(folderNum)
    if not itemFolder then return nil end

    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end

    local closestItem = nil
    local shortestDistance = math.huge

    for _, obj in ipairs(itemFolder:GetChildren()) do
        if obj.Name == "SpawnedItem" or obj:FindFirstChild("SpawnedItem") then
            local item = obj.Name == "SpawnedItem" and obj or obj:FindFirstChild("SpawnedItem")
            
            -- Kiểm tra Attribute Rarity
            local itemRarity = item:GetAttribute("Rarity")
            local isMatch = false
            
            if itemRarity then
                if tostring(itemRarity) == targetRarity or string.find(tostring(itemRarity), targetRarity) then
                    isMatch = true
                end
            else
                -- Nếu không set Attribute Rarity thì mặc định coi như đúng folder
                isMatch = true
            end

            if isMatch then
                local itemPos = item:GetPivot().Position
                local dist = (hrp.Position - itemPos).Magnitude
                if dist < shortestDistance then
                    shortestDistance = dist
                    closestItem = item
                end
            end
        end
    end

    return closestItem
end

local function runQuestLoop()
    if isQuestRunning then return end
    isQuestRunning = true
    
    while autoQuestEnabled do
        -- Bước 1: TP tới NPC Quest
        local npc = workspace:FindFirstChild("Quest") and workspace.Quest:FindFirstChild("NPC")
        if npc then
            teleportTo(npc:GetPivot().Position)
            task.wait(0.3)
            interactPrompt(npc)
            
            -- Confirm Start
            pcall(function()
                ReplicatedStorage.Events.Quest.QuestConfirmStart:FireServer()
            end)
            task.wait(0.5)
        end
        
        -- Bước 2: Nhận diện Rarity và tìm Item gần nhất (3 lượt)
        local targetRarity, targetFolderNum = getNotificationInfo()
        
        if targetRarity and targetFolderNum then
            for step = 1, 3 do
                local spawnedItem = getNearestMatchingItem(targetRarity, targetFolderNum)
                
                if spawnedItem then
                    -- TP đến Item gần nhất & Interact
                    teleportTo(spawnedItem:GetPivot().Position)
                    task.wait(0.3)
                    interactPrompt(spawnedItem)
                    task.wait(0.3)
                    
                    -- TP về NPC & Interact
                    if npc then
                        teleportTo(npc:GetPivot().Position)
                        task.wait(0.3)
                        interactPrompt(npc)
                        task.wait(0.3)
                    end
                end
            end
            
            -- Kết thúc 3 lượt: Decline Reward Event
            pcall(function()
                ReplicatedStorage.Events.Quest.QuestDeclineReward:FireServer()
            end)
            task.wait(0.5)
        else
            task.wait(1)
        end
    end
    
    isQuestRunning = false
end

-- Nút Bật / Tắt Auto Quest
autoQuestBtn.MouseButton1Click:Connect(function()
    autoQuestEnabled = not autoQuestEnabled
    if autoQuestEnabled then
        autoQuestBtn.Text = "Auto Quest: ON"
        autoQuestBtn.BackgroundColor3 = Color3.fromRGB(50, 180, 50)
        task.spawn(runQuestLoop)
    else
        autoQuestBtn.Text = "Auto Quest: OFF"
        autoQuestBtn.BackgroundColor3 = Color3.fromRGB(180, 50, 50)
    end
end)

-- Animation Thu gọn / Mở rộng UI (- / +)
local isMinimized = false
local expandedSize = UDim2.new(0, 180, 0, 155)
local minimizedSize = UDim2.new(0, 180, 0, 30)
local tweenInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

toggleBtn.MouseButton1Click:Connect(function()
    isMinimized = not isMinimized
    if isMinimized then
        toggleBtn.Text = "+"
        scrollFrame.Visible = false
        TweenService:Create(mainFrame, tweenInfo, {Size = minimizedSize}):Play()
    else
        toggleBtn.Text = "-"
        local tween = TweenService:Create(mainFrame, tweenInfo, {Size = expandedSize})
        tween:Play()
        tween.Completed:Connect(function()
            if not isMinimized then
                scrollFrame.Visible = true
            end
        end)
    end
end)
