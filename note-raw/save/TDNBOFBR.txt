-- Tạo các nút Teleport
local function teleportTo(position)
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if hrp then
        hrp.CFrame = CFrame.new(position + Vector3.new(0, 3, 0)) -- Cộng nhẹ Y để không bị kẹt dưới đất
    end
end

for i, loc in ipairs(teleLocations) do
    local btn = Instance.new("TextButton")
    btn.Name = "TeleBtn_" .. i
    btn.Size = UDim2.new(1, -6, 0, 32)
    btn.BackgroundColor3 = Color3.fromRGB(40, 42, 52)
    btn.Text = loc.name
    btn.TextColor3 = Color3.fromRGB(220, 220, 220)
    btn.TextSize = 11
    btn.Font = Enum.Font.GothamMedium
    btn.Parent = scrollFrame

    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 6)
    btnCorner.Parent = btn

    btn.MouseButton1Click:Connect(function()
        teleportTo(loc.pos)
    end)
end

-- Logic Toggle (- / +) với animation mượt
local isMinimized = false
local expandedSize = UDim2.new(0, 180, 0, 155)
local minimizedSize = UDim2.new(0, 180, 0, 30)
local tweenInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

toggleBtn.MouseButton1Click:Connect(function()
    isMinimized = not isMinimized
    if isMinimized then
        toggleBtn.Text = "+"
        scrollFrame.Visible = false
        TweenService:Create(mainFrame, tweenInfo, {Size = minimizedSize}):Play()
    else
        toggleBtn.Text = "-"
        local tween = TweenService:Create(mainFrame, tweenInfo, {Size = expandedSize})
        tween:Play()
        tween.Completed:Connect(function()
            if not isMinimized then
                scrollFrame.Visible = true
            end
        end)
    end
end)
