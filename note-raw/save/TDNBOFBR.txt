---------------------------------------------------------
-- HOOK INSTANT PROMPT (Bấm tức thì, giữ nguyên UI Mobile)
---------------------------------------------------------
local function setupPrompt(prompt)
    if prompt:IsA("ProximityPrompt") then
        prompt.HoldDuration = 0
        prompt.MaxActivationDistance = math.huge
        prompt.RequiresLineOfSight = false
    end
end

-- Hook các prompt hiện có và prompt mới xuất hiện sau này
for _, v in ipairs(workspace:GetDescendants()) do
    setupPrompt(v)
end
workspace.DescendantAdded:Connect(setupPrompt)

    local char = LocalPlayer.Character
    if char and char:FindFirstChild("HumanoidRootPart") then
        char.HumanoidRootPart.CFrame = CFrame.new(pos)
    end

    isMinimized = not isMinimized
    toggleBtn.Text = isMinimized and "+" or "-"
    scrollFrame.Visible = not isMinimized

end)
